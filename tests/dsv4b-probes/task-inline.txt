def truncate(s: str, limit: int = 20) -> str:
    """Return s shortened to at most `limit` terminal columns.
    If it must be shortened, the last kept column is replaced by the ellipsis '…'."""
    if len(s) <= limit:
        return s
    return s[: limit - 1] + "…"
